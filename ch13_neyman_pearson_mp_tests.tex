\chapter{The Neyman--Pearson Lemma and Most Powerful Tests}\label{ch:neyman-pearson}\chaptermark{The Neyman--Pearson Lemma}

Estimation answers the question ``what is $\theta$?''; hypothesis testing\index{hypothesis test@hypothesis test} answers the question ``are the data consistent with a particular claim about $\theta$?''. The contribution of Neyman and Pearson in 1933 \cite{neyman-pearson33} was to set up the testing problem as an optimisation problem: bound the probability of a type I\index{type i and type ii errors@type I and type II errors} error at a prespecified level $\alpha$ and, subject to this constraint, minimise the type II error (maximise the power). For two simple hypotheses the solution is surprisingly simple: reject $H_0$ where the likelihood ratio\index{likelihood ratio@likelihood ratio} is large. At the end of the chapter we discuss the sequential probability ratio\index{sequential probability ratio test sprt@sequential probability ratio test (SPRT)} test, in which observations are collected one at a time and sampling stops once the data speak clearly enough. This chapter lays the foundation for the next three (UMP\index{uniformly most powerful ump test@uniformly most powerful (UMP) test} tests, likelihood ratio tests\index{likelihood ratio test lrt@likelihood ratio test (LRT)}, confidence intervals).

\section{Definitions}

\begin{definition}[Hypotheses]
Let $\Theta=\Theta_0\cup\Theta_1$ be a disjoint decomposition. $H_0:\theta\in\Theta_0$ is the \emph{null hypothesis\index{null hypothesis@null hypothesis}} and $H_1:\theta\in\Theta_1$ the \emph{alternative hypothesis}. If $\Theta_j$ consists of a single point the hypothesis is \emph{simple}, otherwise it is \emph{composite}.
\end{definition}

\begin{definition}[Test Function]
A \emph{(randomised) test} is a measurable function $\phi:\mathcal X^n\to[0,1]$: when $\x$ is observed, $H_0$ is rejected with probability $\phi(\x)$. If $\phi\in\{0,1\}$ the test is \emph{non-randomised} and $\mathcal R=\{\x:\phi(\x)=1\}$ is its \emph{rejection region}.
\end{definition}

\begin{definition}[Error Types and the Power Function]
\[
\beta_\phi(\theta)=\E_\theta[\phi(\X)]\qquad(\text{power function}).
\]
For $\theta\in\Theta_0$, $\beta_\phi(\theta)$ is the probability of a \emph{type I\index{type i and type ii errors@type I and type II errors} error}; for $\theta\in\Theta_1$, $1-\beta_\phi(\theta)$ is the probability of a \emph{type II error} and $\beta_\phi(\theta)$ is the \emph{power} of the test at $\theta$.
\end{definition}

\begin{definition}[Level and Size]
$\sup_{\theta\in\Theta_0}\beta_\phi(\theta)$ is called the \emph{size} of the test. A test with size $\le\alpha$ is called a \emph{level-$\alpha$} test.
\end{definition}

\begin{definition}[Most Powerful (MP) Test]
For simple hypotheses $H_0:\theta=\theta_0$, $H_1:\theta=\theta_1$, a level-$\alpha$ test $\phi^*$ is \emph{most powerful} (MP) if $\beta_{\phi^*}(\theta_1)\ge\beta_\phi(\theta_1)$ for every level-$\alpha$ test $\phi$.
\end{definition}

\begin{definition}[$p$-Value]\label{def:p-degeri}
Let $\{\mathcal R_\alpha\}_{\alpha\in(0,1)}$ be nested rejection regions satisfying $\alpha'<\alpha\implies\mathcal R_{\alpha'}\subseteq\mathcal R_\alpha$. The value $p(\x)=\inf\{\alpha:\x\in\mathcal R_\alpha\}$ is called the \emph{$p$-value}: the smallest level at which $H_0$ can be rejected with the observed data. For tests of the form ``reject for large $T$'', $p(\x)=\sup_{\theta\in\Theta_0}\P_\theta(T(\X)\ge T(\x))$.
\end{definition}

\section{The Neyman--Pearson Lemma}

\begin{theorem}[Neyman--Pearson Lemma \cite{neyman-pearson33}]\label{thm:neyman-pearson}\index{neyman pearson lemma@Neyman--Pearson lemma}
Let $H_0: \theta = \theta_0$, $H_1: \theta = \theta_1$ and $f_j(\x)=L(\theta_j;\x)$.
\begin{enumerate}
  \item \textbf{(Sufficiency)} If for some $k\ge0$ and $\gamma\in[0,1]$
  \[
  \phi^*(\x)=\begin{cases}1, & f_1(\x)>kf_0(\x)\\ \gamma, & f_1(\x)=kf_0(\x)\\ 0, & f_1(\x)<kf_0(\x)\end{cases}\qquad\text{and}\qquad\E_{\theta_0}[\phi^*(\X)]=\alpha,
  \]
  then $\phi^*$ is a most powerful (MP) level-$\alpha$ test.
  \item \textbf{(Existence)} For every $\alpha\in(0,1)$ there is a test of this form with size exactly $\alpha$.
  \item \textbf{(Necessity)} If $\phi$ is any MP level-$\alpha$ test, then $\phi=\phi^*$ a.e.\ on the set $\{f_1\ne kf_0\}$.
\end{enumerate}
Randomisation is needed only for discrete distributions; for continuous distributions $\P_{\theta_0}(f_1=kf_0)=0$ and the test reduces to $\phi^*(\x) = \I\{L(\theta_1)/L(\theta_0) > k\}$.
\end{theorem}
\begin{proof}
(1) Let $\phi$ be any level-$\alpha$ test. For every $\x$,
\[
\big(\phi^*(\x)-\phi(\x)\big)\big(f_1(\x)-kf_0(\x)\big)\ge0,
\]
because when $f_1>kf_0$, $\phi^*=1\ge\phi$; when $f_1<kf_0$, $\phi^*=0\le\phi$; and in the case of equality the second factor is zero. Integrating,
\[
\beta_{\phi^*}(\theta_1)-\beta_\phi(\theta_1)\ge k\big(\beta_{\phi^*}(\theta_0)-\beta_\phi(\theta_0)\big)=k\big(\alpha-\beta_\phi(\theta_0)\big)\ge0 .
\]
(2) Let $R(\x)=f_1(\x)/f_0(\x)$ (on $f_0>0$) and $a(c)=\P_{\theta_0}(R>c)$. $a$ is a non-increasing, right-continuous function of $c$ (the complement of a CDF). Choosing $k=\inf\{c: a(c)\le\alpha\}$ gives $a(k)\le\alpha\le a(k^-)=\P_{\theta_0}(R\ge k)$. If $\P_{\theta_0}(R=k)>0$ take $\gamma=\frac{\alpha-a(k)}{\P_{\theta_0}(R=k)}\in[0,1]$, otherwise $\gamma=0$; the size is then exactly $\alpha$.
(3) If $\phi$ is MP, the inequality in (1) must be an equality; a non-negative integrand with zero integral is zero a.e., i.e.\ $\phi=\phi^*$ wherever $f_1\ne kf_0$.
\end{proof}

\begin{corollary}[An MP Test Is Unbiased]\label{cor:mp-yansiz}
If $\phi^*$ is an MP test\index{most powerful mp test@most powerful (MP) test} of level $\alpha\in(0,1)$, then $\beta_{\phi^*}(\theta_1)\ge\alpha$; if $f_0\neq f_1$, the inequality is strict.
\end{corollary}
\begin{proof}
The test $\phi\equiv\alpha$, which does not use the data at all, has level $\alpha$ and power $\alpha$; the MP test must be at least as powerful.
\end{proof}

\begin{corollary}[Expression via a Sufficient Statistic]\label{cor:np-yeterli}
If $T$ is sufficient and $g(t;\theta_j)$ is the factor in the factorisation\index{factorisation theorem@factorisation theorem}, then the MP test has the form $g(T;\theta_1)/g(T;\theta_0)>k$; that is, it depends only on $T$.
\end{corollary}
\begin{proof}
By factorisation, $f(\x;\theta_j)=g(T(\x);\theta_j)h(\x)$. For every $\x$ with $h(\x)>0$,
$\frac{f(\x;\theta_1)}{f(\x;\theta_0)}=\frac{g(T(\x);\theta_1)}{g(T(\x);\theta_0)}$; the points with $h(\x)=0$ have probability zero under both hypotheses, and how the test is defined there affects neither its level nor its power. Apply Theorem~\ref{thm:neyman-pearson}.
\end{proof}

\begin{theorem}[Distribution of the $p$-Value]\label{thm:p-uniform}
If the distribution of $T$ under (simple) $H_0$ is continuous and the test has the form ``reject for large $T$'', then $p(\X)=1-F_0(T(\X))\sim\text{Uniform}(0,1)$ under $H_0$. For discrete $T$, $\P_{\theta_0}(p(\X)\le u)\le u$ (stochastically larger than uniform).
\end{theorem}
\begin{proof}
By the probability integral transform\index{probability integral transform@probability integral transform} (Theorem~\ref{thm:pit}), $F_0(T)\sim\text{Uniform}(0,1)$, and therefore so is $1-F_0(T)$.
\end{proof}

\section{The Sequential Probability Ratio Test}

The Neyman--Pearson\index{neyman pearson lemma@Neyman--Pearson lemma} lemma fixes the sample size in advance. But if observations arrive one at a time (quality inspection, clinical trials), it is natural to stop once the data speak clearly enough. In the \emph{sequential probability ratio test}, which Wald \cite{wald45} developed for production inspection during the Second World War, one of three decisions is made after each observation: accept $H_0$, reject $H_0$, or take one more observation.

\begin{definition}[SPRT]\label{def:sprt}\index{sequential probability ratio test sprt@sequential probability ratio test (SPRT)|textbf}
Let $X_1,X_2,\dots$ be i.i.d., $H_0:f=f_0$ and $H_1:f=f_1$. Define $Z_i=\ln\frac{f_1(X_i)}{f_0(X_i)}$ and $S_n=\sum_{i=1}^nZ_i$ (the log-likelihood\index{log likelihood@log-likelihood} ratio\index{likelihood ratio@likelihood ratio}). For constants $a<0<b$, with
\[
N=\min\{n\ge1:\ S_n\notin(a,b)\},
\]
$H_0$ is rejected if $S_N\ge b$ and accepted if $S_N\le a$. The event $\{N=n\}$ depends only on $X_1,\dots,X_n$; such a random variable is called a \emph{stopping time\index{stopping time@stopping time}}.
\end{definition}

\begin{lemma}[The SPRT Stops in Finite Time]\label{lem:sprt-durur}\index{sequential probability ratio test sprt@sequential probability ratio test (SPRT)}
If $\P(Z_1=0)<1$, then for some $r\in\N$ and $\delta>0$, $\P(N>n)\le(1-\delta)^{\lfloor n/r\rfloor}$ for every $n$. Hence $N<\infty$ a.s.\ and $\E N^k<\infty$ for every $k$. If $f_0\ne f_1$, the condition holds under both $H_0$ and $H_1$.
\end{lemma}
\begin{proof}
Assume $\P(Z>0)>0$ (the other case is symmetric). Then there is $\varepsilon>0$ with $q=\P(Z>\varepsilon)>0$; choose $r$ with $r\varepsilon>b-a$ and set $\delta=q^r$. Split the observations into consecutive disjoint blocks of $r$. If all $r$ increments of a block exceed $\varepsilon$, the block sum exceeds $b-a$. If $S\in(a,b)$ at the start of the block, then $S$ has left the strip by the end of it. So for $N>kr$ this event must fail in each of the first $k$ blocks; since the blocks are independent, $\P(N>kr)\le(1-\delta)^k$. The geometric tail gives finiteness of all moments. If $f_0\ne f_1$ and $\P_0(Z=0)=1$, then $f_1=f_0$ everywhere on the set where $f_0$ is positive (outside a null set); since the integral of $f_1$ over this set would then be $1$, we would get $f_1=f_0$. The same argument works under $H_1$.
\end{proof}

\begin{theorem}[Wald's Inequalities]\label{thm:wald-sinirlar}
Let $\alpha=\P_0(S_N\ge b)$ and $\beta=\P_1(S_N\le a)$. If $f_0\ne f_1$, then
\[
\alpha\le e^{-b}(1-\beta),\qquad\beta\le e^{a}(1-\alpha).
\]
If, for target error probabilities $\alpha^*,\beta^*$, one chooses $b=\ln\frac{1-\beta^*}{\alpha^*}$ and $a=\ln\frac{\beta^*}{1-\alpha^*}$, then $\alpha\le\frac{\alpha^*}{1-\beta^*}$, $\beta\le\frac{\beta^*}{1-\alpha^*}$ and $\alpha+\beta\le\alpha^*+\beta^*$.
\end{theorem}
\begin{proof}
Let $R_n=\{(x_1,\dots,x_n):N=n,\ S_n\ge b\}$. On $R_n$, $\prod_{i\le n}f_1(x_i)\ge e^b\prod_{i\le n}f_0(x_i)$. By Lemma~\ref{lem:sprt-durur}, $N$ is a.s.\ finite under $H_1$ as well; hence
\[
\alpha=\sum_{n\ge1}\int_{R_n}\prod_{i\le n}f_0\le e^{-b}\sum_{n\ge1}\int_{R_n}\prod_{i\le n}f_1=e^{-b}\,\P_1(S_N\ge b)=e^{-b}(1-\beta).
\]
On the acceptance regions $\{N=n,\ S_n\le a\}$, $\prod f_1\le e^a\prod f_0$, so in the same way $\beta\le e^a(1-\alpha)$. Substituting the chosen $a,b$ gives $\alpha(1-\beta^*)\le\alpha^*(1-\beta)\le\alpha^*$ and $\beta(1-\alpha^*)\le\beta^*(1-\alpha)\le\beta^*$. Adding the middle forms of the first two inequalities and rearranging gives $\alpha+\beta\le\alpha^*+\beta^*$.
\end{proof}

\begin{theorem}[Wald's Identity]\label{thm:wald-ozdeslik}
If $Z_1,Z_2,\dots$ are i.i.d.\ with $\E|Z_1|<\infty$ and $N$ is a stopping time with $\E N<\infty$, then $\E S_N=\E N\cdot\E Z_1$.
\end{theorem}
\begin{proof}
$S_N=\sum_{i\ge1}Z_i\,\I\{N\ge i\}$. The event $\{N\ge i\}=\{N\le i-1\}^c$ depends only on $Z_1,\dots,Z_{i-1}$ and is therefore independent of $Z_i$. First with absolute values: $\E\sum_i|Z_i|\I\{N\ge i\}=\sum_i\E|Z_i|\P(N\ge i)=\E|Z_1|\,\E N<\infty$. This allows the sum and the expectation to be interchanged (Fubini): $\E S_N=\sum_i\E Z_i\,\P(N\ge i)=\E Z_1\,\E N$.
\end{proof}

\begin{corollary}[Average Sample Number]\label{cor:sprt-asn}
Let $f_0\ne f_1$ and let $Z$ be integrable. Since $\E_0Z=-\mathrm{KL}(f_0\|f_1)<0$ and $\E_1Z=\mathrm{KL}(f_1\|f_0)>0$, $\E_jN=\E_jS_N/\E_jZ$ ($j=0,1$). If the overshoot of $S_N$ beyond the boundary is ignored ($S_N\approx b$ or $a$),
\[
\E_0N\approx\frac{\alpha b+(1-\alpha)a}{-\mathrm{KL}(f_0\|f_1)},\qquad\E_1N\approx\frac{(1-\beta)b+\beta a}{\mathrm{KL}(f_1\|f_0)} .
\]
\end{corollary}
\begin{proof}
Lemma~\ref{lem:sprt-durur} gives $\E_jN<\infty$, and Theorem~\ref{thm:wald-ozdeslik} applies. The KL expressions follow from the definition; for their signs see Lemma~\ref{lem:bilgi-esitsizligi}. The approximation replaces the conditional means by the boundaries in the decomposition $\E_0S_N=\alpha\E_0[S_N\mid\text{reject}]+(1-\alpha)\E_0[S_N\mid\text{accept}]$.
\end{proof}

\begin{remark}[Optimality and Optional Stopping]\label{rem:sprt-optimal}
The Wald--Wolfowitz theorem \cite{wald-wolfowitz48} says that among all tests (fixed-sample or sequential) with error probabilities at most $\alpha$ and $\beta$, the SPRT minimises both $\E_0N$ and $\E_1N$ at the same time. The saving is typically of the order of half the fixed sample size (Example~\ref{ex:sprt-normal}). The boundaries of the SPRT account for looking at the data sequentially. By contrast, applying a fixed-sample test at level $\alpha$ after each observation and stopping at the first significant result inflates the type I\index{type i and type ii errors@type I and type II errors} error greatly (Exercise~\ref{exr:13-6}). The group sequential designs used in clinical trials are applications of this idea in which the data are examined in groups.
\end{remark}

\section{Worked Examples}

\begin{example}[Normal Mean: MP Test and the Power Formula]\label{ex:normal-mp}
$X_i\iid\Normal(\mu,\sigma^2)$ with $\sigma$ known; $H_0:\mu=\mu_0$, $H_1:\mu=\mu_1>\mu_0$. The likelihood ratio\index{likelihood ratio@likelihood ratio}
\[
\frac{L(\mu_1)}{L(\mu_0)}=\exp\Big\{\frac{\mu_1-\mu_0}{\sigma^2}\sum x_i-\frac{n(\mu_1^2-\mu_0^2)}{2\sigma^2}\Big\}
\]
is an increasing function of $\bar x$. Hence the MP test\index{most powerful mp test@most powerful (MP) test} has the form $\bar X>c$, and $\P_{\mu_0}(\bar X>c)=\alpha$ gives $c=\mu_0+z_\alpha\sigma/\sqrt n$. Power:
\[
\beta(\mu_1)=\P_{\mu_1}(\bar X>c)=1-\Phi\Big(z_\alpha-\frac{\sqrt n(\mu_1-\mu_0)}{\sigma}\Big).
\]
With $\mu_0=0$, $\mu_1=1$, $\sigma=2$, $n=16$, $\alpha=0.05$: $c=1.645\cdot0.5=0.822$ and $\beta=1-\Phi(1.645-2)=\Phi(0.355)\approx0.639$.

Note: the critical value $c$ \emph{does not depend on $\mu_1$}. This observation is the key to the UMP\index{uniformly most powerful ump test@uniformly most powerful (UMP) test} tests of Chapter~\ref{ch:karlin-rubin}.
\end{example}

\begin{example}[Sample Size Calculation]\label{ex:orneklem-guc}
In the same problem, the $n$ required for a level-$\alpha$ test with power $1-\beta$ is
\[
z_\alpha-\frac{\sqrt n\,\Delta}{\sigma}=-z_\beta\implies n=\Big(\frac{(z_\alpha+z_\beta)\sigma}{\Delta}\Big)^2,\qquad\Delta=\mu_1-\mu_0 .
\]
For $\sigma=2$, $\Delta=1$, $\alpha=0.05$ and power $0.80$ ($z_{0.20}=0.842$), $n=(2.487\cdot2)^2\approx24.7$, i.e.\ $n=25$.
\end{example}

\begin{example}[Poisson: A Randomised MP Test]\label{ex:poisson-rastgele}
$X_1,\dots,X_{10}\iid\text{Poisson}(\lambda)$, $H_0:\lambda=1$, $H_1:\lambda=2$, $\alpha=0.05$. $T=\sum X_i$ is sufficient and the likelihood ratio $(2/1)^Te^{-10}$ is increasing in $T$. Under $H_0$, $T\sim\text{Poisson}(10)$. Since $\P_0(T\ge16)\approx0.0487<0.05<\P_0(T\ge15)\approx0.0835$ (values verified with the code below), no non-randomised test\index{randomised test@randomised test} can have size exactly $0.05$. The MP test is
\[
\phi^*(T)=\begin{cases}1, & T\ge16\\ \gamma, & T=15\\ 0,& T\le14\end{cases},\qquad\gamma=\frac{0.05-\P_0(T\ge16)}{\P_0(T=15)}.
\]
\end{example}

\begin{example}[Normal Variance]
$X_i\iid\Normal(0,\sigma^2)$, $H_0:\sigma^2=\sigma_0^2$, $H_1:\sigma^2=\sigma_1^2>\sigma_0^2$. The likelihood ratio is increasing in $\sum x_i^2$; since $\sum X_i^2/\sigma_0^2\sim\chi^2_n$ under $H_0$, the MP test rejects when $\sum X_i^2>\sigma_0^2\chi^2_{n,\alpha}$.
\end{example}

\begin{example}[SPRT for a Normal Mean]\label{ex:sprt-normal}
$X_i\iid\Normal(\mu,1)$, $H_0:\mu=0$, $H_1:\mu=1$, $\alpha^*=\beta^*=0.05$. $Z_i=X_i-\frac12$, so $S_n=\sum X_i-\frac n2$. The boundaries are $b=\ln19\approx2.944$ and $a=-2.944$. Since $\mathrm{KL}(f_0\|f_1)=\mathrm{KL}(f_1\|f_0)=\frac12$, Corollary~\ref{cor:sprt-asn} gives
\[
\E_0N\approx\frac{0.05(2.944)+0.95(-2.944)}{-0.5}\approx5.3,
\]
and by symmetry $\E_1N\approx5.3$. To achieve the same error probabilities with a fixed sample, the formula of Example~\ref{ex:orneklem-guc} gives $n=\big((z_{0.05}+z_{0.05})/1\big)^2\approx10.8$, that is, $11$ observations. On average the SPRT\index{sequential probability ratio test sprt@sequential probability ratio test (SPRT)} makes do with about half. Figure~\ref{fig:sprt} shows three sample paths.
\end{example}

\subsection{The Sequential Probability Ratio Test (SPRT)}
The SPRT\index{sequential probability ratio test sprt@sequential probability ratio test (SPRT)} of Example~\ref{ex:sprt-normal} is run $10^5$ times under each hypothesis. The actual error probabilities are compared with the bounds of Theorem~\ref{thm:wald-sinirlar}, and the average sample number with the approximation of Corollary~\ref{cor:sprt-asn}. The last line is the fixed sample size giving the same error probabilities.

\pycode{code/ch13/c13_4_sequential_probability_ratio_test.py}
\begin{lstlisting}[style=output]
boundaries: a = -2.944, b = 2.944
H0: type I error = 0.0290 (target 0.05, Wald bound alpha/(1-beta) = 0.0526)
    E[N] = 6.93 | Wald approximation (ignoring overshoot) = 5.30 | median N = 6 | P(N > 11) = 0.137 | max N = 60
H1: type II error = 0.0289 (target 0.05, Wald bound beta/(1-alpha) = 0.0526)
    E[N] = 6.91 | Wald approximation (ignoring overshoot) = 5.30 | median N = 6 | P(N > 11) = 0.137 | max N = 54
n required by the fixed-sample test = 11
\end{lstlisting}

The actual error probabilities ($\approx0.029$) are below the target ($0.05$) and satisfy Wald's bounds. The reason is the overshoot: since the test stops by passing the boundary rather than exactly on it, the likelihood ratio\index{likelihood ratio@likelihood ratio} decides on stronger evidence than required. For the same reason the average sample number ($6.9$) is above Wald's approximation ($5.3$). Still, it is on average $37\%$ fewer than the $11$ observations the fixed-sample test needs. The price is variability: about $14\%$ of the runs need more than 11 observations, and the longest run took 60. In practice, therefore, truncated SPRTs with an upper limit on the number of observations are used.

\section{Comments and Notes}

\begin{remark}[An Asymmetric Approach]
The Neyman--Pearson\index{neyman pearson lemma@Neyman--Pearson lemma} framework does not treat $H_0$ and $H_1$ symmetrically: the type I\index{type i and type ii errors@type I and type II errors} error is kept under control, while the type II error is only minimised. Therefore ``$H_0$ could not be rejected'' does not mean ``$H_0$ is true''; a low-power test frequently fails to reject $H_0$. This is why power analysis is an integral part of experimental design.
\end{remark}

\begin{remark}[Randomisation in Practice]
Randomised tests\index{randomised test@randomised test} are needed in theory (for the existence of an MP test\index{most powerful mp test@most powerful (MP) test}), but they are rarely used in practice, because the same data can lead to different decisions. Practitioners report either the conservative test ($\gamma=0$) or the $p$-value; the ``mid-$p$'' value $\P(T>t)+\frac12\P(T=t)$ is a compromise.
\end{remark}

\begin{remark}[What Does a $p$-Value Not Measure?]
The $p$-value is \emph{not} the probability that $H_0$ is true; it is the probability, when $H_0$ is true, of obtaining a result at least as extreme as the one observed. Nor does it measure effect size: for very large $n$, even negligibly small differences produce very small $p$-values. The 2016 statement of the American Statistical Association \cite{wasserstein16} discusses these misinterpretations in detail. The relationship between $p$-values and posterior probabilities is taken up in Chapter~\ref{ch:bayes}.
\end{remark}

\section{Chapter Summary}
\begin{summary}
\begin{itemize}
  \item A test is a decision rule. The probability of a type I error is bounded by the level, and the power is maximised subject to this constraint.
  \item The Neyman--Pearson lemma (Theorem~\ref{thm:neyman-pearson}): for simple hypotheses the most powerful test is the likelihood ratio test. For discrete distributions, randomisation may be needed to attain the level exactly.
  \item An MP test is unbiased (Corollary~\ref{cor:mp-yansiz}) and depends only on a sufficient statistic (Corollary~\ref{cor:np-yeterli}).
  \item The $p$-value (Definition~\ref{def:p-degeri}) is the probability under $H_0$ of a result at least as extreme as the one observed; for a continuous test statistic it is uniformly distributed under $H_0$ (Theorem~\ref{thm:p-uniform}). The $p$-value is not the probability that $H_0$ is true, and it does not measure effect size.
  \item The SPRT collects observations until the log-likelihood ratio leaves the strip $(a,b)$. Its error probabilities are bounded by Wald's inequalities (Theorem~\ref{thm:wald-sinirlar}), and its average sample number follows from Wald's identity (Theorem~\ref{thm:wald-ozdeslik}). For the same errors, the SPRT needs about half as many observations as the fixed-sample test.
\end{itemize}
\end{summary}
